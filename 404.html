path = decodeURIComponent(location.pathname)

siteIndex = fetch("/assets/data/site-index.json").then(r => r.json())

norm = s => s.toLowerCase().replace(/\.html?$/, "").replace(/\/index$/, "").replace(/^\/+|\/+$/g, "")
words = s => norm(s).split(/[^a-z0-9]+/).filter(w => w && w !== "index" && w !== "www")
lev = (a, b) => {
  const m = Array.from({length: a.length + 1}, (_, i) => [i]);
  for (let j = 1; j <= b.length; j++) m[0][j] = j;
  for (let i = 1; i <= a.length; i++)
    for (let j = 1; j <= b.length; j++)
      m[i][j] = Math.min(m[i-1][j] + 1, m[i][j-1] + 1, m[i-1][j-1] + (a[i-1] === b[j-1] ? 0 : 1));
  return m[a.length][b.length];
}
sim = (a, b) => (a || b) ? 1 - lev(a, b) / Math.max(a.length, b.length) : 0
simk = (g, k) => g.length >= 4 && k.includes(g) ? Math.max(sim(g, k), 0.85) : sim(g, k)

scored = {
  const segs = norm(path).split("/").filter(Boolean);
  const cands = segs.filter(g => g.length >= 4).length ? segs.filter(g => g.length >= 4) : segs;
  const first = segs[0] || "";
  const wanted = new Set(words(path));
  const rows = siteIndex.map(d => {
    const keys = [norm(d.title), norm(d.id), norm(new URL(d.url, location.origin).pathname).split("/").pop()];
    const s = Math.max(...cands.flatMap(g => keys.map(k => simk(g, k) * (g === cands[cands.length - 1] ? 1 : 0.65))), 0);
    const dw = new Set([...words(d.title), ...words(d.id)]);
    const overlap = [...wanted].filter(w => dw.has(w)).length / (new Set([...wanted, ...dw]).size || 1);
    const bonus = first && d.url.startsWith("/" + first + "/") ? 0.15 : 0;
    const isParent = d.url !== "/" && path.startsWith(d.url) ? -0.08 : 0;
    return {...d, score: 0.65 * s + 0.2 * overlap + bonus + isParent};
  }).sort((a, b) => b.score - a.score).slice(0, 5);
  const u = rows.map(r => 7 * r.score), mx = Math.max(...u);
  const den = u.reduce((t, x) => t + Math.exp(x - mx), 0);
  return rows.map((r, i) => ({...r, p: Math.exp(u[i] - mx) / den}));
}

stationOf = d => {
  const byUrl = {"/": "Home", "/explore/": "Explore", "/reports/": "Reports", "/data/": "Data", "/code/": "Code",
    "/updates/": "Updates", "/about.html": "About", "/contact/": "About", "/license/": "About"};
  if (d.kind === "Page") return d.url.startsWith("/models/") ? "Models" : (byUrl[d.url] || "Home");
  if (["App", "Map"].includes(d.kind)) return "Explore";
  if (d.kind === "Data") return "Data";
  if (d.kind === "Model") return "Models";
  if (["Estimation", "Calibration", "Validation", "Input data", "Post-processing", "Scenario", "Geography", "Development", "Analysis"].includes(d.kind)) return "Reports";
  return "Code";
}
